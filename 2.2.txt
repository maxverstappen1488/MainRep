#define _USE_MATH_DEFINES
#include <iostream>
#include <cmath>
using namespace std;

/**
 * @brief - вычисляет значение кусочно-заданной функции y(x)
 * @param x - входное значение аргумента
 * @return - вычисленное значение функции y
 * 
 * @details Функция вычисляет значение по формуле:
 * - если x < 0.5: y = π/2 + x·a - e^(a·x), где a = 1.36
 * - если x ≥ 0.5: y = x³·a + 28
 */
double sys(double x) {
    double a = 1.36;
    double y;
    if (x < 0.5) {
        y = M_PI / 2 + x * a - exp(a * x);
    }
    else {
        y = pow(x, 3) * a + 28;
    }
    return y;
}

/**
 * @brief - точка входа в программу
 * @return 0, если программа выполнена корректно
 * 
 * @details Программа запрашивает у пользователя значение аргумента x,
 * вычисляет значение функции sys(x) и выводит результат.
 */
int main() {
    // настройка локали для поддержки русского языка
    setlocale(LC_ALL, "RU");
    
    // ввод
    double x;
    cout << "Введите значение x: ";
    cin >> x;
    
    // вычисление
    double y = sys(x);
    
    // вывод результата
    cout << "y = " << y << endl;
    
    return 0;
}
