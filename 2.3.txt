#include <iostream>
#include <cstdlib>
using namespace std;

/**
 * @brief - проверяет корректность ввода положительного числа
 * @return - валидное положительное число типа double
 *
 * @details Функция считывает строку ввода, преобразует её в число
 * с помощью atof(). Если результат <= 0, запрашивает ввод повторно.
 * Возвращает только положительное число.
 */
double check();

/**
 * @brief - проверяет возможность размещения двух домов на участке
 * @param a - ширина участка
 * @param b - высота участка
 * @param p - первая сторона первого дома
 * @param q - вторая сторона первого дома
 * @param r - первая сторона второго дома
 * @param s - вторая сторона второго дома
 * @return - 1.0, если размещение возможно; 0.0, если невозможно
 *
 * @details Функция перебирает все 4 комбинации ориентаций домов
 * (каждый дом можно повернуть на 90°). Проверяет два варианта размещения:
 * - горизонтальное: дома стоят рядом по ширине (w1 + w2 <= a),
 *   высота ограничена максимальным из h1, h2
 * - вертикальное: дома стоят друг над другом по высоте (h1 + h2 <= b),
 *   ширина ограничена максимальным из w1, w2
 */
double willBuild(double a, double b, double p, double q, double r, double s);

/**
 * @brief - точка входа в программу
 * @return 0, если программа выполнена корректно
 *
 * @details Программа запрашивает у пользователя:
 * - размеры участка (a, b)
 * - размеры первого дома (p, q)
 * - размеры второго дома (r, s)
 * Все значения проходят валидацию через функцию check().
 * Затем вызывается willBuild() для проверки возможности размещения.
 * Результат выводится в виде "Да/Нет".
 */
int main() {
    // настройка локали для поддержки русского языка
    setlocale(LC_ALL, "RU");

    double a, b, p, q, r, s;

    // ввод размеров участка и домов с проверкой
    cout << "Введите значение переменной a: ";
    a = check();
    cout << "Введите значение переменной b: ";
    b = check();
    cout << "Введите значение переменной p: ";
    p = check();
    cout << "Введите значение переменной q: ";
    q = check();
    cout << "Введите значение переменной r: ";
    r = check();
    cout << "Введите значение переменной s: ";
    s = check();

    // проверка возможности размещения и вывод результата
    if (willBuild(a, b, p, q, r, s))
        cout << "Да, дома можно разместить.\n";
    else
        cout << "Нет, дома не помещаются.\n";

    return 0;
}

double check() {
    char Input[100]; // каждый элемент списка - отдельный символ
    double result;
    cin >> Input;
    result = atof(Input); // объединяет цифры в "число"; возвращает 0, если ввод некорректный
    while (result <= 0) {
        cout << "Значение " << Input << " введено неверно." << endl;
        cin >> Input;
        result = atof(Input); // объединяет цифры в "число"; возвращает 0, если ввод некорректный
    }
    return result;
}

double willBuild(double a, double b, double p, double q, double r, double s) {
    bool possible = false;
    // Перебор всех 4 комбинаций ориентаций
    for (int i = 0; i < 2; ++i) {
        for (int j = 0; j < 2; ++j) {
            double w1 = (i == 0) ? p : q;
            double h1 = (i == 0) ? q : p;
            double w2 = (j == 0) ? r : s;
            double h2 = (j == 0) ? s : r;
            // Горизонтальное размещение
            if (w1 + w2 <= a && max(h1, h2) <= b) {
                possible = true;
                break; // прерываем, если возможно
            }
            // Вертикальное размещение
            if (max(w1, w2) <= a && h1 + h2 <= b) {
                possible = true;
                break; // прерываем, если возможно
            }
        }
        if (possible == true) {
            break;
        }
    }
    return possible;
}
